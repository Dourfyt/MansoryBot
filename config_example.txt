# Конфигурация задержек для планировщика
# Скопируйте эти переменные в .env файл и настройте значения под ваши нужды

# Задержки между отправками стикеров (в секундах)
STICKER_DELAY_BASE=0.5        # Базовая задержка между стикерами
STICKER_DELAY_10=2.0          # Задержка каждые 10 чатов
STICKER_DELAY_50=5.0          # Задержка каждые 50 чатов

# Задержки между отправкой информации (в секундах)
INFO_DELAY_BASE=0.3           # Базовая задержка между информацией
INFO_DELAY_20=2.0             # Задержка каждые 20 чатов

# Задержки между сбросами баз данных (в секундах)
RESET_DELAY_BASE=0.1          # Базовая задержка между сбросами
RESET_DELAY_50=1.0            # Задержка каждые 50 баз

# Количество повторных попыток
MAX_RETRY_ATTEMPTS=3

# Задержки при ошибках (в секундах)
ERROR_DELAY_SHORT=5            # Короткая задержка при ошибке
ERROR_DELAY_LONG=30            # Длинная задержка при превышении лимитов

# Рекомендации по настройке:
# - Для очень большого количества чатов (1000+) увеличьте задержки:
#   STICKER_DELAY_BASE=1.0
#   STICKER_DELAY_10=5.0
#   STICKER_DELAY_50=10.0
#
# - При частых ошибках "Too Many Requests" увеличьте ERROR_DELAY_LONG до 60-120
# - При стабильной работе можно уменьшить задержки для ускорения процесса
